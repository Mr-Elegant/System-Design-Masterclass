Problem 1: Design Tic-Tac-Toe (O(1) Win Condition)

OOPTic-Tac-ToeBoard State

Design a Tic-Tac-Toe game supporting $N \times N$ board size with $O(1)$ win-condition evaluation.

JavaScript — Tic-Tac-Toe Engine
class TicTacToeEngine {
    #size; #rows; #cols; #diag = 0; #antiDiag = 0;
    constructor(size = 3) {
        this.#size = size;
        this.#rows = Array(size).fill(0);
        this.#cols = Array(size).fill(0);
    }
    makeMove(r, c, symbol) {
        const val = symbol === 'X' ? 1 : -1;
        this.#rows[r] += val; this.#cols[c] += val;
        if (r === c) this.#diag += val;
        if (r + c === this.#size - 1) this.#antiDiag += val;

        if (Math.abs(this.#rows[r]) === this.#size || Math.abs(this.#cols[c]) === this.#size ||
            Math.abs(this.#diag) === this.#size || Math.abs(this.#antiDiag) === this.#size) {
            return "WIN";
        }
        return "CONTINUE";
    }
}
const game = new TicTacToeEngine(3);
console.log(game.makeMove(0, 0, 'X'));

Problem 2: Design Vending Machine (State Pattern & Refunds)

State PatternVending MachineRefunds

Complete Vending Machine handling Idle, Coin Insertion, Product Selection, Dispensing, and Refund Cancellation using the State Pattern.

JavaScript — Vending Machine State Pattern
class State {
    insertCoin(vm, amt) { throw new Error("Action invalid in current state"); }
    selectProduct(vm, code) { throw new Error("Action invalid in current state"); }
    dispense(vm) { throw new Error("Action invalid in current state"); }
    refund(vm) { throw new Error("Action invalid in current state"); }
}

class IdleState extends State {
    insertCoin(vm, amt) {
        vm.balance += amt;
        console.log(`[IDLE -> HAS_MONEY] Coin Inserted: $${amt} | Total: $${vm.balance}`);
        vm.setState(vm.hasMoneyState);
    }
}

class HasMoneyState extends State {
    selectProduct(vm, code) {
        const item = vm.inventory.get(code);
        if (!item || item.count <= 0) throw new Error("Item out of stock!");
        if (vm.balance < item.price) throw new Error("Insufficient funds!");
        vm.selectedProduct = item;
        console.log(`[HAS_MONEY -> DISPENSING] Selected: ${item.name}`);
        vm.setState(vm.dispensingState);
    }
    refund(vm) {
        console.log(`[REFUND] Returned $${vm.balance}`);
        vm.balance = 0;
        vm.setState(vm.idleState);
    }
}

class DispensingState extends State {
    dispense(vm) {
        const item = vm.selectedProduct;
        item.count--;
        vm.balance -= item.price;
        console.log(`🎁 [DISPENSED] ${item.name}! Change: $${vm.balance}`);
        vm.balance = 0;
        vm.selectedProduct = null;
        vm.setState(vm.idleState);
    }
}

class VendingMachine {
    constructor() {
        this.idleState = new IdleState();
        this.hasMoneyState = new HasMoneyState();
        this.dispensingState = new DispensingState();
        this.currentState = this.idleState;
        this.balance = 0;
        this.selectedProduct = null;
        this.inventory = new Map([["A1", { name: "Coke", price: 2.50, count: 5 }]]);
    }
    setState(s) { this.currentState = s; }
    insertCoin(amt) { this.currentState.insertCoin(this, amt); }
    selectProduct(code) { this.currentState.selectProduct(this, code); }
    dispense() { this.currentState.dispense(this); }
    refund() { this.currentState.refund(this); }
}

const vm = new VendingMachine();
vm.insertCoin(3.00);
vm.selectProduct("A1");
vm.dispense();

Problem 4: Design Library Management System (Modelling & Loans)

ModellingLibraryCatalog

Library management system supporting ISBN catalog search, member loan checkouts, fine calculation, and reservation queues.

JavaScript — Library Management System
class BookItem {
    constructor(isbn, title, author) {
        this.isbn = isbn;
        this.title = title;
        this.author = author;
        this.isBorrowed = false;
        this.dueDate = null;
    }
}

class LibraryCatalog {
    #books = new Map(); // ISBN -> BookItem
    #loans = new Map(); // LoanId -> Loan Record

    addBook(book) { this.#books.set(book.isbn, book); }

    checkoutBook(isbn, memberId, durationDays = 14) {
        const book = this.#books.get(isbn);
        if (!book) throw new Error("Book not found!");
        if (book.isBorrowed) throw new Error("Book is already checked out!");

        book.isBorrowed = true;
        book.dueDate = Date.now() + durationDays * 86400000;

        const loan = { loanId: `LOAN_${Date.now()}`, isbn, memberId, dueDate: book.dueDate };
        this.#loans.set(loan.loanId, loan);
        console.log(`📚 [CHECKOUT] Book '${book.title}' checked out by Member ${memberId}`);
        return loan;
    }

    returnBook(isbn, memberId) {
        const book = this.#books.get(isbn);
        if (!book || !book.isBorrowed) throw new Error("Invalid return request!");

        const overDueDays = Math.max(0, Math.ceil((Date.now() - book.dueDate) / 86400000));
        const fine = overDueDays * 1.5; // $1.50 per day late fine

        book.isBorrowed = false;
        book.dueDate = null;
        console.log(`📖 [RETURN] Book '${book.title}' returned. Late Fine: $${fine}`);
        return fine;
    }
}

const lib = new LibraryCatalog();
lib.addBook(new BookItem("ISBN-101", "Grokking System Design", "DesignGurus"));
const loan = lib.checkoutBook("ISBN-101", "Member_77");
lib.returnBook("ISBN-101", "Member_77");

Problem 6: Design Snake & Ladder (Turn Queue & Board Jumps)

OOPSnake & LadderTurn Queue

Snake & Ladder board game with dynamic snakes/ladders jump map, dice rolling, and circular player turn queue.

JavaScript — Snake & Ladder Game Engine
class SnakeLadderGame {
    #size = 100;
    #jumps = new Map(); // startPos -> endPos (Snakes & Ladders)
    #players = [];
    #turnIndex = 0;

    constructor(players) {
        this.#players = players.map(p => ({ name: p, pos: 0 }));
        // Add Ladders (start < end)
        this.#jumps.set(2, 38); this.#jumps.set(7, 14); this.#jumps.set(15, 26);
        // Add Snakes (start > end)
        this.#jumps.set(99, 54); this.#jumps.set(70, 55); this.#jumps.set(52, 29);
    }

    rollDice() { return Math.floor(Math.random() * 6) + 1; }

    playTurn() {
        const player = this.#players[this.#turnIndex];
        const dice = this.rollDice();
        let nextPos = player.pos + dice;

        if (nextPos > this.#size) {
            console.log(`🎲 ${player.name} rolled ${dice}, but needs exact roll to reach 100.`);
        } else {
            // Apply Snake/Ladder jump if present
            if (this.#jumps.has(nextPos)) {
                const destination = this.#jumps.get(nextPos);
                const type = destination > nextPos ? "LADDER 🪜" : "SNAKE 🐍";
                console.log(`🎲 ${player.name} landed on ${nextPos} -> Hit a ${type}! Moved to ${destination}`);
                nextPos = destination;
            } else {
                console.log(`🎲 ${player.name} rolled ${dice} -> Moved from ${player.pos} to ${nextPos}`);
            }
            player.pos = nextPos;
        }

        if (player.pos === this.#size) {
            console.log(`🏆 [WINNER] ${player.name} won the game!`);
            return "WIN";
        }

        this.#turnIndex = (this.#turnIndex + 1) % this.#players.length;
        return "CONTINUE";
    }
}

const game = new SnakeLadderGame(["Alice", "Bob"]);
game.playTurn();

Problem 7: Design Chess (Piece Hierarchy & Command Pattern Undo)

Complex ModellingChessCommand Pattern

Complete Chess game with Piece hierarchy (King, Queen, Rook, Bishop, Knight, Pawn), 8x8 matrix board, move validation, and Command pattern for Undo/Redo.

JavaScript — Chess Game Engine
class Position { constructor(r, c) { this.r = r; this.c = c; } }

class Piece {
    constructor(color, pos) { this.color = color; this.pos = pos; }
    getValidMoves(board) { throw new Error("Abstract method"); }
}

class Rook extends Piece {
    getValidMoves(board) {
        const moves = [];
        const dirs = [[1,0], [-1,0], [0,1], [0,-1]];
        for (const [dr, dc] of dirs) {
            let r = this.pos.r + dr, c = this.pos.c + dc;
            while (r >= 0 && r < 8 && c >= 0 && c < 8) {
                moves.push(new Position(r, c));
                r += dr; c += dc;
            }
        }
        return moves;
    }
}

class MoveCommand {
    constructor(board, piece, fromPos, toPos) {
        this.board = board; this.piece = piece; this.fromPos = fromPos; this.toPos = toPos;
        this.capturedPiece = null;
    }

    execute() {
        this.capturedPiece = this.board.getPiece(this.toPos);
        this.board.setPiece(this.toPos, this.piece);
        this.board.setPiece(this.fromPos, null);
        this.piece.pos = this.toPos;
    }

    undo() {
        this.board.setPiece(this.fromPos, this.piece);
        this.board.setPiece(this.toPos, this.capturedPiece);
        this.piece.pos = this.fromPos;
    }
}

class ChessBoard {
    #grid = Array(8).fill(null).map(() => Array(8).fill(null));
    getPiece(pos) { return this.#grid[pos.r][pos.c]; }
    setPiece(pos, p) { this.#grid[pos.r][pos.c] = p; }
}

const board = new ChessBoard();
const rook = new Rook("WHITE", new Position(0, 0));
board.setPiece(new Position(0,0), rook);
const cmd = new MoveCommand(board, rook, new Position(0,0), new Position(0,5));
cmd.execute(); console.log("Rook moved to (0,5)");
cmd.undo(); console.log("Move undone! Rook back at (0,0)");

Problem 9: Design ATM System (State Pattern & Cash Dispenser)

StateATMGreedy Dispenser

ATM System with PIN verification, card validation, and greedy Cash Dispenser algorithm ($100, $50, $20 bills).

JavaScript — ATM Cash Dispenser Engine
class CashDispenser {
    #inventory = new Map([[100, 10], [50, 20], [20, 50]]); // Denomination -> Count

    dispense(amount) {
        if (amount % 10 !== 0) throw new Error("Withdrawal amount must be multiple of $10");

        const result = new Map();
        let remaining = amount;
        const denominations = [100, 50, 20];

        for (const denom of denominations) {
            const available = this.#inventory.get(denom) || 0;
            const countNeeded = Math.floor(remaining / denom);
            const countToTake = Math.min(available, countNeeded);

            if (countToTake > 0) {
                result.set(denom, countToTake);
                remaining -= countToTake * denom;
            }
        }

        if (remaining > 0) throw new Error("ATM cannot dispense exact cash denomination!");

        // Commit inventory deduction
        result.forEach((count, denom) => {
            this.#inventory.set(denom, this.#inventory.get(denom) - count);
        });

        console.log(`💵 [CASH DISPENSED] Total: $${amount} | Bills:`, Object.fromEntries(result));
        return result;
    }
}

const atm = new CashDispenser();
atm.dispense(270); // 2x $100, 1x $50, 1x $20

Problem 10: Design Coffee Machine (Factory & Strategy Pattern)

FactoryCoffee MachineStrategy

Automated Coffee Machine preparing Espresso, Cappuccino, and Latte with ingredient inventory validation.

JavaScript — Coffee Machine Factory
class CoffeeMachine {
    #inventory = { water: 500, milk: 500, beans: 200 };

    brew(recipeName) {
        const recipes = {
            Espresso: { water: 50, milk: 0, beans: 18 },
            Latte: { water: 50, milk: 150, beans: 18 },
            Cappuccino: { water: 50, milk: 100, beans: 18 }
        };

        const recipe = recipes[recipeName];
        if (!recipe) throw new Error(`Unknown coffee recipe: ${recipeName}`);

        if (this.#inventory.water < recipe.water ||
            this.#inventory.milk < recipe.milk ||
            this.#inventory.beans < recipe.beans) {
            throw new Error(`Insufficient ingredients to brew ${recipeName}!`);
        }

        this.#inventory.water -= recipe.water;
        this.#inventory.milk -= recipe.milk;
        this.#inventory.beans -= recipe.beans;

        console.log(`☕ [BREWED] ${recipeName} ready! Remaining Inventory:`, this.#inventory);
        return recipeName;
    }
}

const cm = new CoffeeMachine();
cm.brew("Latte");
