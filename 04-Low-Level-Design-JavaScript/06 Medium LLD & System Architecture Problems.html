Problem 8: Design Splitwise (Strategy & Debt Graph Minimization)

StrategySplitwiseGreedy Debt Graph

Expense sharing system (Splitwise) supporting Equal, Exact, and Percentage splits. Minimizes total settlements between $N$ users via a Greedy Debt Minimization Graph Algorithm.

  ┌───────────────────────────────────────────────────────────┐
  │              SPLITWISE DEBT SIMPLIFICATION                │
  ├───────────────────────────────────────────────────────────┤
  │ Initial Expenses:                                         │
  │   Alice paid $90 for Alice, Bob, Charlie ($30 each)      │
  │   Bob owes Alice $30, Charlie owes Alice $30              │
  │                                                           │
  │ Net Balance Calculation:                                  │
  │   Alice:   +$60 (Creditor)                                │
  │   Bob:     -$30 (Debtor)                                  │
  │   Charlie: -$30 (Debtor)                                  │
  │                                                           │
  │ Greedy Settlement Matches:                                │
  │   Transaction 1: Bob pays Alice $30                       │
  │   Transaction 2: Charlie pays Alice $30                   │
  └───────────────────────────────────────────────────────────┘
JavaScript — Splitwise Debt Graph Simplification
class SplitwiseEngine {
    #balances = new Map(); // UserID -> Net Balance (+ Creditor, - Debtor)

    // Add expense record and calculate net balance changes
    addExpense(payerId, totalAmount, splits) {
        let splitSum = 0;
        splits.forEach(s => splitSum += s.amount);
        if (Math.abs(splitSum - totalAmount) > 0.01) {
            throw new Error("Split amounts do not equal total expense amount!");
        }

        // Payer gets credited total amount paid
        this.#balances.set(payerId, (this.#balances.get(payerId) || 0) + totalAmount);

        // Participants get debited their respective split amounts
        splits.forEach(s => {
            this.#balances.set(s.userId, (this.#balances.get(s.userId) || 0) - s.amount);
        });
    }

    // Greedy Min-Transaction Graph Minimization Algorithm O(N log N)
    simplifyDebts() {
        const debtors = [];
        const creditors = [];

        // Separate users into debtors and creditors
        this.#balances.forEach((bal, userId) => {
            if (bal < -0.01) debtors.push({ userId, amount: -bal });
            else if (bal > 0.01) creditors.push({ userId, amount: bal });
        });

        const transactions = [];
        let i = 0, j = 0;

        // Greedy matching loop
        while (i < debtors.length && j < creditors.length) {
            const debtor = debtors[i];
            const creditor = creditors[j];
            const settleAmount = Math.min(debtor.amount, creditor.amount);

            transactions.push({
                from: debtor.userId,
                to: creditor.userId,
                amount: Number(settleAmount.toFixed(2))
            });

            debtor.amount -= settleAmount;
            creditor.amount -= settleAmount;

            if (debtor.amount < 0.01) i++;
            if (creditor.amount < 0.01) j++;
        }
        return transactions;
    }
}

const engine = new SplitwiseEngine();
engine.addExpense("Alice", 90, [
    { userId: "Alice", amount: 30 },
    { userId: "Bob", amount: 30 },
    { userId: "Charlie", amount: 30 }
]);
console.log("Settlement Transactions:", engine.simplifyDebts());

Problem 14: Design Meeting Scheduler (Interval Overlap & Conflicts)

ConflictsMeeting SchedulerInterval Overlap

Meeting Scheduler detecting calendar conflict overlaps, finding available slots across $N$ participants, and booking rooms.

  ┌───────────────────────────────────────────────────────────┐
  │                 MEETING SCHEDULER DATAFLOW                 │
  ├───────────────────────────────────────────────────────────┤
  │ User A Schedule: [09:00 - 10:00], [13:00 - 14:00]        │
  │ User B Schedule: [09:30 - 11:00], [15:00 - 16:00]        │
  │ Requested Meeting: [09:30 - 10:30]                        │
  │                                                           │
  │ Conflict Check:                                           │
  │   User A [09:00 - 10:00] OVERLAPS [09:30 - 10:30] -> CONFLICT!│
  │   User B [09:30 - 11:00] OVERLAPS [09:30 - 10:30] -> CONFLICT!│
  │                                                           │
  │ Free Slot Finder:                                         │
  │   Available Joint Free Slot: [11:00 - 12:00]             │
  └───────────────────────────────────────────────────────────┘
JavaScript — Meeting Scheduler Implementation
class Interval {
    constructor(start, end) {
        if (start >= end) throw new Error("Start time must be before end time");
        this.start = start; // Epoch timestamp or minutes from midnight
        this.end = end;
    }

    overlaps(other) {
        return Math.max(this.start, other.start) < Math.min(this.end, other.end);
    }
}

class UserCalendar {
    #userId; #meetings = [];

    constructor(userId) { this.#userId = userId; }

    hasConflict(newInterval) {
        return this.#meetings.some(m => m.interval.overlaps(newInterval));
    }

    addMeeting(meeting) {
        if (this.hasConflict(meeting.interval)) {
            throw new Error(`Calendar conflict for user ${this.#userId}`);
        }
        this.#meetings.push(meeting);
        this.#meetings.sort((a, b) => a.interval.start - b.interval.start);
    }

    getMeetings() { return [...this.#meetings]; }
}

class MeetingScheduler {
    #calendars = new Map();

    registerUser(userId) {
        if (!this.#calendars.has(userId)) {
            this.#calendars.set(userId, new UserCalendar(userId));
        }
    }

    bookMeeting(organizerId, participantIds, interval, title) {
        const allUsers = [organizerId, ...participantIds];

        // Validate no user has a schedule conflict
        for (const uid of allUsers) {
            const cal = this.#calendars.get(uid);
            if (!cal) throw new Error(`User ${uid} not registered!`);
            if (cal.hasConflict(interval)) {
                console.log(`❌ Meeting Booking Failed! User ${uid} has a schedule conflict.`);
                return false;
            }
        }

        const meeting = { id: `M_${Date.now()}`, title, interval, organizerId, participantIds };

        // Commit meeting to all participant calendars
        for (const uid of allUsers) {
            this.#calendars.get(uid).addMeeting(meeting);
        }

        console.log(`✅ Meeting '${title}' successfully booked for ${allUsers.length} participants!`);
        return meeting;
    }
}

const scheduler = new MeetingScheduler();
scheduler.registerUser("Alice");
scheduler.registerUser("Bob");

const slot1 = new Interval(540, 600); // 09:00 - 10:00 (minutes from midnight)
scheduler.bookMeeting("Alice", ["Bob"], slot1, "Sprint Planning");

const slot2 = new Interval(570, 630); // 09:30 - 10:30 (Conflict!)
scheduler.bookMeeting("Alice", ["Bob"], slot2, "Design Review");

Problem 15: Design BookMyShow (Seat Locking & Concurrency)

ConcurrencyBookMyShowRedis 10-Min Lock

Movie ticket booking engine preventing double-booking via 10-minute temporary seat locks powered by Redis TTL keys.

  ┌───────────────────────────────────────────────────────────┐
  │                BOOKMYSHOW SEAT LOCKING FLOW               │
  ├───────────────────────────────────────────────────────────┤
  │ User Selects Seats [A1, A2]                               │
  │   │                                                       │
  │   ▼                                                       │
  │ Try Acquire Redis Key: SET lock:show1:A1 userId NX EX 600 │
  │   │                                                       │
  │   ├───► Key Already Exists -> Seat Locked by Another User! │
  │   │                                                       │
  │   └───► Lock Acquired (10-Min TTL Timer Started)           │
  │           │                                               │
  │           ▼                                               │
  │      User Completes Payment inside 10 Mins                │
  │           │                                               │
  │           ├───► Payment Success -> Confirm Booking in DB  │
  │           └───► Payment Timeout -> Redis Lock Expires (Free)│
  └───────────────────────────────────────────────────────────┘
JavaScript — BookMyShow Concurrent Seat Booking
class BookMyShowEngine {
    #seatLocks = new Map(); // Key: `${showId}_${seatId}` -> { userId, expiresAt }
    #confirmedBookings = new Set(); // Confirmed Seats

    lockSeats(showId, seatIds, userId, ttlMs = 600000) { // Default 10-minute TTL
        const now = Date.now();

        // Step 1: Check availability and current locks
        for (const seatId of seatIds) {
            const key = `${showId}_${seatId}`;
            if (this.#confirmedBookings.has(key)) {
                throw new Error(`Seat ${seatId} is already permanently booked!`);
            }

            const lock = this.#seatLocks.get(key);
            if (lock && lock.expiresAt > now && lock.userId !== userId) {
                throw new Error(`Seat ${seatId} is temporarily locked by another user!`);
            }
        }

        // Step 2: Acquire temporary locks
        const expiresAt = now + ttlMs;
        for (const seatId of seatIds) {
            const key = `${showId}_${seatId}`;
            this.#seatLocks.set(key, { userId, expiresAt });
        }

        console.log(`🔒 [SEATS LOCKED] Seats [${seatIds.join(', ')}] locked for user ${userId} for 10 minutes.`);
        return true;
    }

    confirmBooking(showId, seatIds, userId) {
        const now = Date.now();

        // Validate user holds valid unexpired lock
        for (const seatId of seatIds) {
            const key = `${showId}_${seatId}`;
            const lock = this.#seatLocks.get(key);
            if (!lock || lock.userId !== userId || lock.expiresAt <= now) {
                throw new Error(`Seat lock expired or invalid for seat ${seatId}!`);
            }
        }

        // Confirm booking permanently
        for (const seatId of seatIds) {
            const key = `${showId}_${seatId}`;
            this.#confirmedBookings.add(key);
            this.#seatLocks.delete(key); // Release lock
        }

        console.log(`🎟️ [BOOKING CONFIRMED] Seats [${seatIds.join(', ')}] permanently booked for user ${userId}!`);
        return true;
    }
}

const bms = new BookMyShowEngine();
bms.lockSeats("SHOW_101", ["A1", "A2"], "User_1");
bms.confirmBooking("SHOW_101", ["A1", "A2"], "User_1");

Problem 24: Design Ride Sharing System (Uber - Complex Domain)

Complex DomainUberDriver Matching

Ride-sharing system matching riders with nearby drivers, spatial location tracking, and trip state machines.

JavaScript — Ride Sharing Trip Engine
const TripStatus = Object.freeze({ REQUESTED: "REQUESTED", ACCEPTED: "ACCEPTED", IN_TRIP: "IN_TRIP", COMPLETED: "COMPLETED" });

class RideSharingSystem {
    #drivers = new Map(); // driverId -> { location, isAvailable }
    #trips = new Map();

    registerDriver(driverId, lat, lng) {
        this.#drivers.set(driverId, { driverId, lat, lng, isAvailable: true });
    }

    requestRide(riderId, pickupLat, pickupLng, dropLat, dropLng) {
        // Find nearest available driver using Euclidean distance
        let nearestDriver = null;
        let minDistance = Infinity;

        this.#drivers.forEach(driver => {
            if (driver.isAvailable) {
                const dist = Math.hypot(driver.lat - pickupLat, driver.lng - pickupLng);
                if (dist < minDistance) {
                    minDistance = dist;
                    nearestDriver = driver;
                }
            }
        });

        if (!nearestDriver) throw new Error("No available drivers nearby!");

        nearestDriver.isAvailable = false; // Reserve driver
        const trip = {
            tripId: `TRIP_${Date.now()}`,
            riderId,
            driverId: nearestDriver.driverId,
            status: TripStatus.ACCEPTED,
            fare: Number((minDistance * 10 + 5).toFixed(2))
        };

        this.#trips.set(trip.tripId, trip);
        console.log(`🚕 [RIDE MATCHED] Rider ${riderId} matched with Driver ${nearestDriver.driverId} | Fare: $${trip.fare}`);
        return trip;
    }
}

const uber = new RideSharingSystem();
uber.registerDriver("Driver_John", 37.7749, -122.4194);
uber.requestRide("Rider_Alice", 37.7750, -122.4195, 37.7800, -122.4200);

Problem 25: Design Food Delivery System (UberEats - Complex Domain)

Complex DomainUberEatsOrder State Machine

Food delivery system handling restaurant menus, cart checkout, driver pickup routing, and order status state machine.

JavaScript — Food Delivery Order Lifecycle
const OrderStatus = Object.freeze({ PLACED: "PLACED", PREPARING: "PREPARING", OUT_FOR_DELIVERY: "OUT_FOR_DELIVERY", DELIVERED: "DELIVERED" });

class FoodDeliveryEngine {
    #orders = new Map();

    createOrder(customerId, restaurantId, items) {
        let total = 0;
        items.forEach(i => total += i.price * i.qty);

        const order = {
            orderId: `ORD_${Date.now()}`,
            customerId,
            restaurantId,
            items,
            total,
            status: OrderStatus.PLACED
        };

        this.#orders.set(order.orderId, order);
        console.log(`🍱 [ORDER PLACED] Order ${order.orderId} placed at Restaurant ${restaurantId} | Total: $${total}`);
        return order;
    }

    updateOrderStatus(orderId, newStatus) {
        const order = this.#orders.get(orderId);
        if (!order) throw new Error("Order not found!");
        order.status = newStatus;
        console.log(`🚚 [ORDER UPDATE] Order ${orderId} status changed to: ${newStatus}`);
        return order;
    }
}

const eats = new FoodDeliveryEngine();
const order = eats.createOrder("Cust_1", "Rest_BurgerKing", [{ name: "Whopper", price: 8.99, qty: 2 }]);
eats.updateOrderStatus(order.orderId, OrderStatus.PREPARING);
