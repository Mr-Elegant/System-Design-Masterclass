// Define Immutable Enums for Spot and Vehicle Classifications
const SpotType = Object.freeze({ SMALL: "SMALL", MEDIUM: "MEDIUM", LARGE: "LARGE", EV: "EV" });
const VehicleType = Object.freeze({ MOTORCYCLE: "MOTORCYCLE", CAR: "CAR", TRUCK: "TRUCK", EV_CAR: "EV_CAR" });

class ParkingSpot {
    // Encapsulated Private Member Fields
    #id;               // Unique Spot Identifier (e.g. "EV_FLOOR1_101")
    #floor;            // Floor number where spot is located
    #type;             // Spot classification from SpotType enum
    #isOccupied = false; // Flag tracking availability
    #currentVehicle = null; // Reference to parked vehicle instance

    constructor(id, floor, type) {
        this.#id = id;
        this.#floor = floor;
        this.#type = type;
    }

    // Evaluate compatibility between vehicle type and spot classification
    canFit(vehicle) {
        if (this.#isOccupied) return false; // Occupied spots cannot accept vehicles
        if (vehicle.type === VehicleType.EV_CAR) return this.#type === SpotType.EV; // Require EV charging spot
        if (vehicle.type === VehicleType.MOTORCYCLE) return true; // Motorcycles fit anywhere
        if (vehicle.type === VehicleType.CAR) return this.#type !== SpotType.SMALL; // Cars fit in MEDIUM/LARGE
        return this.#type === SpotType.LARGE; // Trucks require LARGE spots
    }

    // State Mutation: Park vehicle into spot
    park(vehicle) {
        if (!this.canFit(vehicle)) throw new Error(`Spot ${this.#id} cannot fit vehicle: ${vehicle.type}`);
        this.#currentVehicle = vehicle;
        this.#isOccupied = true; // Lock spot availability
    }

    // State Mutation: Free spot upon vehicle exit
    unpark() {
        const releasedVehicle = this.#currentVehicle;
        this.#currentVehicle = null;
        this.#isOccupied = false; // Unlock spot availability
        return releasedVehicle;
    }

    getId() { return this.#id; }
    getOccupiedStatus() { return this.#isOccupied; }
}

class HourlyPricingStrategy {
    // Pricing rate table mapping vehicle type to hourly rate ($/hr)
    #rateMap = { MOTORCYCLE: 5, CAR: 10, TRUCK: 20, EV_CAR: 15 };

    calculateFee(ticket) {
        // Calculate elapsed time in hours (rounded up to nearest full hour)
        const durationMs = Date.now() - ticket.entryTime;
        const hours = Math.max(1, Math.ceil(durationMs / 3600000));
        
        // Return computed fee
        const hourlyRate = this.#rateMap[ticket.vehicle.type] || 10;
        return hours * hourlyRate;
    }
}

class ParkingLotController {
    static #instance = null; // Singleton Instance Reference
    #spots = [];             // Managed spot inventory array
    #activeTickets = new Map(); // Ticket Map: ticketId -> Ticket Object
    #pricingStrategy;

    constructor(pricingStrategy = new HourlyPricingStrategy()) {
        if (ParkingLotController.#instance) return ParkingLotController.#instance; // Enforce Singleton
        this.#pricingStrategy = pricingStrategy;
        ParkingLotController.#instance = this;
    }

    // Register a new spot into lot inventory
    addSpot(spot) {
        this.#spots.push(spot);
    }

    // Gate Operation: Issue ticket and park vehicle
    issueTicket(vehicle) {
        // Search spot inventory for first compatible available spot
        const spot = this.#spots.find(s => s.canFit(vehicle));
        if (!spot) throw new Error(`Parking Lot FULL for vehicle type: ${vehicle.type}`);

        // Park vehicle into selected spot
        spot.park(vehicle);

        // Generate unique ticket record
        const ticket = {
            ticketId: `TICK_${Date.now()}_${Math.floor(Math.random()*1000)}`,
            vehicle,
            spot,
            entryTime: Date.now()
        };

        // Persist ticket into active registry
        this.#activeTickets.set(ticket.ticketId, ticket);
        console.log(`🚗 [PARKED] ${vehicle.licensePlate} (${vehicle.type}) in Spot ${spot.getId()} | Ticket: ${ticket.ticketId}`);
        return ticket;
    }

    // Gate Operation: Process exit, calculate fee, and release spot
    processExit(ticketId) {
        const ticket = this.#activeTickets.get(ticketId);
        if (!ticket) throw new Error("Invalid or Expired Ticket ID!");

        // Unpark vehicle to free spot
        ticket.spot.unpark();

        // Compute total fee via Strategy Pattern
        const fee = this.#pricingStrategy.calculateFee(ticket);

        // Remove ticket from active registry
        this.#activeTickets.delete(ticketId);
        console.log(`💳 [EXIT PROCESSED] Ticket: ${ticketId} | Fee: $${fee}`);
        return fee;
    }
}

const Direction = Object.freeze({ UP: "UP", DOWN: "DOWN", IDLE: "IDLE" });

class ElevatorCar {
    constructor(id) {
        this.id = id;
        this.currentFloor = 1;
        this.direction = Direction.IDLE;
        this.requests = new Set(); // Set storing unique requested target floors
    }

    // Register a new floor destination into request queue
    addFloorRequest(floor) {
        this.requests.add(floor);
    }

    // Step state machine forward by 1 tick unit
    step() {
        if (this.requests.size === 0) {
            this.direction = Direction.IDLE; // Reset to idle when no pending requests
            return;
        }

        // Sort requested floors in ascending numerical order
        const floors = Array.from(this.requests).sort((a, b) => a - b);

        // LOOK/SCAN Algorithm Decision Logic
        if (this.direction === Direction.UP || this.direction === Direction.IDLE) {
            // Find next target floor >= current floor
            const next = floors.find(f => f >= this.currentFloor) || floors[floors.length - 1];
            if (next > this.currentFloor) {
                this.currentFloor++;
                this.direction = Direction.UP;
            } else if (next < this.currentFloor) {
                this.currentFloor--;
                this.direction = Direction.DOWN;
            } else {
                // Arrived at requested floor! Remove request and open doors
                this.requests.delete(this.currentFloor);
                console.log(`🛗 [ELEVATOR ${this.id}] Doors Opened at Floor ${this.currentFloor}`);
            }
        }
    }
}

const el = new ElevatorCar(1);
el.addFloorRequest(5);
el.step(); console.log(`Elevator Current Floor: ${el.currentFloor}`);
